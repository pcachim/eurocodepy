    eurocodepy

    Provide several functions to help designers working with Eurocodes.

    The package provides modules for different Eurocodes and utilities, such as:

    • dbase: access to the database of materials, loads, sections, etc.
    • utils: utility functions for calculations, stress, and section properties.
    • ec1: utility functions for ec1 calculations.
    • ec2: utility functions for ec2 calculations.
    • ec3: utility functions for ec3 calculations.
    • ec5: utility functions for ec5 calculations.
    • ec7: utility functions for ec7 calculations.
    • ec8: utility functions for ec8 calculations.

    """Represents EC2 concrete properties and provides access to parameters.

    Attributes
    ----------
    grade : str
        The concrete grade (e.g., 'C30/37').
    name : str
        The normalized name of the concrete grade.
    fck : float
        Characteristic compressive strength (MPa).
    fcm : float
        Mean compressive strength (MPa).
    fctm : float
        Mean tensile strength (MPa).
    fctk_05 : float
        5% fractile tensile strength (MPa).
    fctk_95 : float
        95% fractile tensile strength (MPa).
    Ecm : float
        Modulus of elasticity (MPa).
    eps_c2 : float
        Strain at peak stress.
    eps_cu2 : float
        Ultimate compressive strain.
    n : float
        Parameter n for stress-strain curve.
    gamma_c : float
        Partial safety factor for concrete.
    fcd : float
        Design compressive strength (MPa).
    fctd : float
        Design tensile strength (MPa).
    weight : float
        Weight of the concrete per cubic meter (kN/m³).

    Methods
    -------
    __repr__()
        Returns a string representation of the Concrete object.
    __str__()
        Returns a user-friendly string representation of the Concrete object.
    from_fck(f_ck: int, name: object = None)
        Class method to create a Concrete instance from characteristic
        compressive strength.

    """

    def __init__(self, class_name: str = "C30/37", edition: str | None = None,
                 situation: str = "persistent", loading: str = "late",
                 k_cip: float = 1.0, annex: str | None = None) -> None:
        """Create a concrete of the given class.

        Parameters
        ----------
        class_name : str
            Strength class, e.g. ``"C30/37"``.
        edition : str, optional
            ``"2004"`` or ``"2023"``; ``None`` is the default edition (2004).
            It selects the grades (EN 1992-1-1:2023 Table 5.1, with C100/115),
            ``Ecm`` and the design values.
        situation : str
            Design situation for the partial factors (EN 1992-1-1:2023
            Table 4.3): ``"persistent"``, ``"fatigue"``, ``"accidental"`` or
            ``"serviceability"``. Only ``"persistent"`` for 2004.
        loading : str
            2023 only, selects ``k_tc`` and ``k_tt`` (5.1.6): ``"late"`` is
            ``t_ref`` <= 28 d (CN/CR) or 56 d (CS) with the design loading not
            expected for 3 months after casting (``k_tc`` = 1,0; ``k_tt`` = 0,8);
            ``"other"`` is every other case, including ``fck(t)``
            (``k_tc`` = 0,85; ``k_tt`` = 0,7).
        k_cip : float
            Factor on ``gamma_c`` for cast-in-place members cast in the ground
            without permanent casing (EN 1992-1-1:2023 4.3.3(4): 1,1 in
            general, 1,0 for EN 1536/1538/14199). Default 1,0.
        annex : str, optional
            Country code of a national annex (see :mod:`eurocodepy.annex`) whose
            values replace the recommended ones; ``None`` is the Eurocode.

        """
        if loading not in LOADING_KEYS:
            msg = f"loading must be one of {tuple(LOADING_KEYS)}, got '{loading}'"
            raise ValueError(msg)
        self.grade = class_name
        self.edition = dbase.resolve_edition("ec2", edition) or "2004"
        self.situation = situation
        self.loading = loading
        self.k_cip = k_cip
        self.annex = annex

        class_name = class_name.replace("/", "_").upper()
        mats = dbase.get_edition_data("ec2", edition, annex)["Concrete"]
        grades = mats["Grade"]
        if class_name not in grades:
            grades_list = [item.replace("_", "/") for item in grades]
            msg = (
                f"Concrete class '{class_name}' not found in database. "
                f"Concrete type must be one of {grades_list}"
            )
            raise ValueError(msg)

        conc = grades[class_name]
        self.name = class_name
        self.fck = conc["fck"]  # MPa
        self.fcm = conc["fcm"]  # Mean compressive strength (MPa)
        self.fctm = conc["fctm"]  # Mean tensile strength (MPa)
        self.fctk_05 = conc["fctk05"]  # 5% fractile (MPa)
        self.fctk_95 = conc.get("fctk95", round(1.3 * self.fctm, 1))  # 95% fractile (MPa)
        self.Ecm = conc["Ecm"]  # Modulus of elasticity (MPa)
        self.eps_c2 = conc["epsc2"]  # Strain at peak stress
        self.eps_cu2 = conc["epscu2"]  # Ultimate compressive strain
        self.n = conc["n"]  # Ultimate compressive strain
        self.weight = Weight  # kN/m³

        self._set_design_values(mats["Parameters"])

    def _set_design_values(self, params: dict) -> None:
        """Set the partial factors, ``eta_cc``, ``k_tc``, ``k_tt``, ``fcd``, ``fctd``.

        2004: ``fcd = fck / gamma_c`` (``alpha_cc`` is not applied here; the
        ULS functions take it as an argument). 2023: Eqs. (5.3)-(5.5),
        ``fcd = eta_cc * k_tc * fck / gamma_c`` and
        ``fctd = k_tt * fctk,0.05 / gamma_c``.
        """
        factors = material_partial_factors(self.edition, self.situation, self.annex)
        self.gamma_c = factors["gamma_c"] * self.k_cip
        self.gamma_ce = factors["gamma_ce"]
        self.gamma_ce_tested = factors["gamma_ce_tested"]
        self.gamma_v = factors["gamma_v"]
        if "k_tc" not in params:  # EN 1992-1-1:2004
            self.eta_cc = 1.0
            self.k_tc = 1.0
            self.k_tt = 1.0
            self.fcd = round(self.fck / self.gamma_c, 1)  # Design strength (MPa)
            self.fctd = round(self.fctk_05 / (params["gamma_ct"] * self.k_cip), 1)
            return
        self.fck_ref = params["fck_ref"]
        self.eta_cc, self.k_tc, self.k_tt = concrete_strength_factors(
            self.fck, self.edition, self.loading, self.annex)  # (5.4)
        self.fcd = self.eta_cc * self.k_tc * self.fck / self.gamma_c  # (5.3)
        self.fctd = self.k_tt * self.fctk_05 / self.gamma_c  # (5.5)

    def __repr__(self) -> str:
        """Return a string representation of the Concrete object.

        Returns
        -------
        str
            String representation of the Concrete object.

        """
        return (
            f"Concrete(grade='{self.grade}', fck = {self.fck}, fcm = {self.fcm}, "
            f"fctm = {self.fctm}, fctk_05 = {self.fctk_05}, fctk_95 = {self.fctk_95}, "
            f"Ecm = {self.Ecm}, eps_c2 = {self.eps_c2}, eps_cu2 = {self.eps_cu2}, "
            f"n = {self.n}, fcd = {self.fcd}, fctd = {self.fctd})"
        )

    def __str__(self) -> str:
        """Return a string representation of the Concrete object.

        Returns
        -------
        str
            String representation of the Concrete object.

        """
        return (
            f"Concrete {self.grade} (\n    "
            f"fck = {self.fck} MPa, \n    "
            f"fcm = {self.fcm} MPa, \n    "
            f"fctm = {self.fctm:.1f} MPa, \n    "
            f"fctk_05 = {self.fctk_05:.1f} MPa, \n    "
            f"fctk_95 = {self.fctk_95:.1f} MPa, \n    "
            f"Ecm = {self.Ecm} MPa, \n    "
            f"eps_c2 = {self.eps_c2}, \n    "
            f"eps_cu2 = {self.eps_cu2}, \n    "
            f"n = {self.n}, \n    "
            f"fcd = {self.fcd} MPa, \n    "
            f"fctd = {self.fctd} MPa\n    "
            f"weight = self.weight kN/m³\n"
            f")"
        )

    @classmethod
    def from_fck(cls, f_ck: int, name: object = None, edition: str | None = None,
                 situation: str = "persistent", loading: str = "late",
                 k_cip: float = 1.0, annex: str | None = None) -> "Concrete":
        """Create a Concrete instance from characteristic compressive strength.

        Args:
            f_ck (int): Characteristic compressive strength of concrete (MPa)
            name (str, optional): Name for the concrete grade.
            edition (str, optional): ``"2004"`` (default) or ``"2023"``.
                For 2023, ``fctm`` follows Table 5.1 and ``Ecm`` Eq. (5.1)
                (``kE`` = 9500) and the parabola-rectangle diagram has ``εc2`` = 2,0 ‰,
                ``εcu`` = 3,5 ‰ and ``n`` = 2 for every class (8.1.2(1)).
            situation, loading, k_cip, annex: see :class:`Concrete`.

        Returns:
            Concrete: Concrete instance.

        """
        concrete = Concrete("C30/37", edition, situation, loading, k_cip, annex)  # Dummy
        fck = float(f_ck)
        concrete.fck = round(fck, 1)  # MPa
        concrete.fcm = round(fck + 8, 1)  # Mean compressive strength (MPa)
        if concrete.edition == "2004":
            concrete.fctm = round(0.30 * fck**(2 / 3), 1)  # Mean tensile strength (MPa)
            concrete.Ecm = round(22000 * (concrete.fcm / 10)**0.3, 1)  # (MPa)
        else:
            concrete.fctm = round(
                0.30 * fck**(2 / 3) if fck <= REF_FCK else 1.1 * fck**(1 / 3), 1)
            concrete.Ecm = round(KE_2023 * concrete.fcm ** (1 / 3), 1)
        concrete.fctk_05 = round(0.7 * concrete.fctm, 1)  # 5% fractile
        concrete.fctk_95 = round(1.3 * concrete.fctm, 1)  # 95% fractile
        if concrete.edition == "2004":
            concrete.eps_c2 = 2.0 if fck <= REF_FCK else 2.0 + 0.085 * (fck - 50) ** 0.53
            # Strain at peak stress
            concrete.eps_cu2 = (
                3.5 if fck <= REF_FCK
                else round(2.6 + 35 * ((90.0 - fck) / 100.0) ** 4, 1)
            )
            concrete.n = (
                2.0
                if fck <= REF_FCK
                else round(1.4 + 23.4 * ((90.0 - fck) / 100.0) ** 4, 1)
            )
        else:                       # EN 1992-1-1:2023 8.1.2(1), Eq. (8.4)
            concrete.eps_c2, concrete.eps_cu2, concrete.n = 2.0, 3.5, 2.0
        if name is not None:
            concrete.name = name
            concrete.grade = name
        else:
            concrete.name = f"C{f_ck}"
            concrete.grade = f"C{f_ck}"

        concrete._set_design_values(
            dbase.get_edition_data("ec2", edition, annex)["Concrete"]["Parameters"])

        return concrete

def from_fck(cls, f_ck: int, name: object = None, edition: str | None = None,
             situation: str = "persistent", loading: str = "late",
             k_cip: float = 1.0, annex: str | None = None) -> "Concrete":
    """Create a Concrete instance from characteristic compressive strength.

    Args:
        f_ck (int): Characteristic compressive strength of concrete (MPa)
        name (str, optional): Name for the concrete grade.
        edition (str, optional): ``"2004"`` (default) or ``"2023"``.
            For 2023, ``fctm`` follows Table 5.1 and ``Ecm`` Eq. (5.1)
            (``kE`` = 9500) and the parabola-rectangle diagram has ``εc2`` = 2,0 ‰,
            ``εcu`` = 3,5 ‰ and ``n`` = 2 for every class (8.1.2(1)).
        situation, loading, k_cip, annex: see :class:`Concrete`.

    Returns:
        Concrete: Concrete instance.

    """
    concrete = Concrete("C30/37", edition, situation, loading, k_cip, annex)  # Dummy
    fck = float(f_ck)
    concrete.fck = round(fck, 1)  # MPa
    concrete.fcm = round(fck + 8, 1)  # Mean compressive strength (MPa)
    if concrete.edition == "2004":
        concrete.fctm = round(0.30 * fck**(2 / 3), 1)  # Mean tensile strength (MPa)
        concrete.Ecm = round(22000 * (concrete.fcm / 10)**0.3, 1)  # (MPa)
    else:
        concrete.fctm = round(
            0.30 * fck**(2 / 3) if fck <= REF_FCK else 1.1 * fck**(1 / 3), 1)
        concrete.Ecm = round(KE_2023 * concrete.fcm ** (1 / 3), 1)
    concrete.fctk_05 = round(0.7 * concrete.fctm, 1)  # 5% fractile
    concrete.fctk_95 = round(1.3 * concrete.fctm, 1)  # 95% fractile
    if concrete.edition == "2004":
        concrete.eps_c2 = 2.0 if fck <= REF_FCK else 2.0 + 0.085 * (fck - 50) ** 0.53
        # Strain at peak stress
        concrete.eps_cu2 = (
            3.5 if fck <= REF_FCK
            else round(2.6 + 35 * ((90.0 - fck) / 100.0) ** 4, 1)
        )
        concrete.n = (
            2.0
            if fck <= REF_FCK
            else round(1.4 + 23.4 * ((90.0 - fck) / 100.0) ** 4, 1)
        )
    else:                       # EN 1992-1-1:2023 8.1.2(1), Eq. (8.4)
        concrete.eps_c2, concrete.eps_cu2, concrete.n = 2.0, 3.5, 2.0
    if name is not None:
        concrete.name = name
        concrete.grade = name
    else:
        concrete.name = f"C{f_ck}"
        concrete.grade = f"C{f_ck}"

    concrete._set_design_values(
        dbase.get_edition_data("ec2", edition, annex)["Concrete"]["Parameters"])

    return concrete

        type_label (str, optional): Timber grade label (e.g., 'C24').

    def __init__(self, type_label: str = "C24",  # noqa: D107
                 edition: str | None = None) -> None:
        tdb = _timber_db(edition)
        self.edition = edition
        self._params = tdb["Parameters"]
        try:
            timber = tdb["Grade"][type_label]
        except KeyError:
            msg = (
                f"Timber type '{type_label}' not found. "
                f"Available types: {list(dbase.TimberGrades)}"
            )
            raise ValueError(
                msg,
            ) from None

        self.type_label = type_label
        self.type = timber["Type"]
        self.material = TimberType.TIMBER
        self.timber_product = TimberProduct.ST

        self.fmk = timber["fmk"]
        self.ft0k = timber["ft0k"]
        self.ft90k = timber["ft90k"]

        self.E90mean = timber["E90mean"]

        self.Gk = self.E0k / self.E0mean * self.Gmean

        self.rhok = timber["rhok"]
        self.rhom = timber["rhom"]

        params = self._params
        self.safety = params["safety"][self.type]
        self.kmods = params["kmod"][self.type]

        self.ft0d = 0.0
        self.fc0d = 0.0

        self.fc90d = 0.0

    @property
    def kcr(self) -> float:
        """Crack factor for shear, ``b_ef = k_cr·b`` (EN 1995-1-1:2004 §6.1.7(2)).

        Defaults to 0.67 for solid timber and glued laminated timber and 1.0
        for the other products; assign a value to override it (e.g. a National
        Annex value). Not used by the 2025 edition, which has no ``b_ef``
        (the cracking effect is in ``k_v``, §8.1.11.1).

            return override
        if self.material in (TimberType.TIMBER, TimberType.GLULAM):

        return 1.0

    def k_mod(
        self,
        service_class: ServiceClass | str = ServiceClass.SC1,
        load_duration: LoadDuration | str = LoadDuration.MediumDuration,
    ) -> float:
        """Return the kmod value for serviceability limit state.

        Args:
            service_class (ServiceClass, optional): Service class (SC1, SC2, or SC3).
                Defaults to ServiceClass.SC1.
            load_duration (LoadDuration, optional): Load duration (Perm, Long,
            Medium, Short, Inst).
                Defaults to LoadDuration.Medium.

        Returns:
            float: kmod value for the specified service class and load duration.

        """
        if isinstance(load_duration, str) and load_duration not in {
            "Perm", "Long", "Medium", "Short", "Inst", "Permanent", "LongDuration",
            "MediumDuration", "ShortDuration", "Instantaneous",
        }:
            load_duration = 0
        if isinstance(load_duration, LoadDuration):
            load_duration = load_duration.value

        if (isinstance(service_class, str) and
            service_class not in {"SC1", "SC2", "SC3", "SC4"}):
            service_class = "SC3"
        if isinstance(service_class, ServiceClass):
            service_class = service_class.name

        try:
            kmod = self.kmods[service_class][load_duration]
        except KeyError:
            msg = (f"No k_mod for {self.type_label} ({self.type}) in "
                   f"{service_class} in EN 1995-1-1"
                   f"{':' + self.edition if self.edition else ''}.")
            raise KeyError(msg) from None
        if kmod is None:
            msg = (f"No k_mod for {self.type_label} ({self.type}) in "
                   f"{service_class} for this load duration.")
            raise KeyError(msg)
        self.kmod = kmod

        return self.kmod

    def design_values(
        self,
        service_class: ServiceClass | str = ServiceClass.SC1,
        load_duration: LoadDuration | str = LoadDuration.MediumDuration,

        ``0.83`` for LVL in SC2/SC3 and GLVL in SC2 (a requirement), and ``1.2``

    def k_h(self, size: float,
            forcetype: TimberForcesType = TimberForcesType.Bending) -> float:
        """Return the kh value for serviceability limit state.

        Returns:
            float: kh value.

        """
        match forcetype:
            case TimberForcesType.Bending:
                aref = self.kh["aref"]
                sm = self.kh["sm"]
                kmin = self.kh["khmmin"]
                kmax = self.kh["khmmax"]
            case TimberForcesType.Shear:
                aref = self.kh["aref"]
                sm = self.kh["sv"]
                kmin = self.kh["khvmin"]
                kmax = self.kh["khvmax"]
            case TimberForcesType.Tension:
                if "st" in self.kh:          # EN 1995-1-1:2025 Table 5.6
                    aref = self.kh["atref"]
                    sm = self.kh["st"]
                    kmin = self.kh["khtmin"]
                    kmax = self.kh["khtmax"]
                else:
                    aref = self.kh["aref"]
                    sm = self.kh["sm"]
                    kmin = self.kh["khmmin"]
                    kmax = self.kh["khmmax"]
            case _:
                return 1.0

        kh = (aref / size)**sm
        return max(kmin, min(kh, kmax))

    def __post_init__(self):

        if self.steel_grade not in dbase.SteelGrades:
            msg = (
                f"Weld grade '{self.steel_grade}' not found in database. "
                f"Weld grade must be one of {list(dbase.SteelGrades.keys())}"
            )
            raise ValueError(msg)

        self.fu = steel["fuk"]
        self.fy = steel["fyk"]

        # Fillet width for tension full strength (should be mukltiplied by t)
        self.a_full_tension = 0.5 * self.fy / self.gamma_M0 / self.fv_wd
        self.fw_vrd = self.fu / self.gamma_M2 / self.beta_w
        self.fw_trd = 0.9 * self.fu / self.gamma_M2

        self.cos = np.cos(self.orientation)
        self.sin = np.sin(self.orientation)
        self.x_max = self.x + 0.5 * self.length * self.cos
        self.y_max = self.y + 0.5 * self.length * self.sin
        self.x_min = self.x - 0.5 * self.length * self.cos
        self.y_min = self.y - 0.5 * self.length * self.sin
        i_xx = (self.a**3 * self.length) / 12.0
        i_yy = (self.a * self.length**3) / 12.0
        i_xy = 0.5 * (i_xx + i_yy)
        i_yx = 0.5 * (i_xx - i_yy)
        cos_2a = np.cos(2 * self.orientation)
        sin_2a = np.sin(2 * self.orientation)
        self.inertia_xx = i_xy + i_yx * cos_2a
        self.inertia_yy = i_xy - i_yx * cos_2a
        self.inertia_xy = i_yx * sin_2a
        self.area = self.a * self.length

        self.area_yy = self.area * np.round(self.sin**2, 2)

    def seismic_get_params(code, seismic_type, soil_type, importance_class) -> dict:
    """Get the seismic spectrum parameters.

    This function retrieves the seismic spectrum parameters from the SeismicLoads
    database based on the provided parameters. It returns a dictionary containing
    the spectrum parameters, including the importance coefficient. The parameters
    are case-insensitive for soil type and importance class, but the seismic type
    should be provided in uppercase. The function is designed to work with different
    localization codes, such as "PT" for Portugal or "EU" for standard CEN. The
    seismic types can include "PT1", "PT2", "PTA" for Portugal, or "CEN-1", "CEN-2"
    for CEN standards. This function is useful for engineers and designers who need
    to apply seismic loads according to Eurocode standards.

    Args:
        code (str): Localization code (e.g., "PT" for Portugal, "EU" for standard
            CEN).
        seismic_type (str): type of seismic action (e.g., "PT1", "PT2", "PTA",
            "CEN-1", "CEN-2").
        soil_type (str): soil type (e.g., "A", "B", "C", "D", "E" for different
            soil conditions).
        importance_class (str): importance class (e.g., "i", "ii", "iii", "iv").

    Returns:
        dict: spectrum parameters for the specified code, seismic type, soil type,
            and importance class.

    Raises:
        ValueError: If the code, seismic type, soil type, or importance class is
            invalid.

    """
    code = str.upper(code)
    seismic_type = str.upper(seismic_type)
    soil_type = str.upper(soil_type)
    importance_class = str.lower(importance_class)
    if code not in SeismicLoads["Locale"]:
        msg = f"Invalid code: {code}. Available codes: {list(SeismicLoads['Locale'])}"
        raise ValueError(msg)
    locale_data = SeismicLoads["Locale"][code]
    if seismic_type not in locale_data["Spectrum"]:
        msg = (f"Invalid seismic type: {seismic_type}. Available types: "
               f"{list(locale_data['Spectrum'])}")
        raise ValueError(msg)
    if soil_type not in locale_data["Spectrum"][seismic_type]:
        msg = (f"Invalid soil type: {soil_type}. Available types: "
               f"{list(locale_data['Spectrum'][seismic_type])}")
        raise ValueError(msg)
    if importance_class not in locale_data["ImportanceClass"]:
        msg = (f"Invalid importance class: {importance_class}. Available classes: "
               f"{list(locale_data['ImportanceClass'])}")
        raise ValueError(msg)

    # A copy: the database is shared, and ImportanceCoef is specific to this call
    spectrum = copy.deepcopy(locale_data["Spectrum"][seismic_type][soil_type])
    spectrum["ImportanceCoef"] = locale_data["ImportanceCoef"][seismic_type][
        importance_class]
    return spectrum

    def wind_get_params(code: str = "PT", zone: str = "ZonaA", terrain: str = "II") -> tuple:
    """Get the wind load parameters for a given code, zone, and terrain.

    This function retrieves the wind load parameters from the WindLoads database based
    on the provided parameters.
    It returns a tuple containing the base velocity, minimum height, roughness length,
    and roughness length for terrain II.
    The parameters are case-insensitive for the zone and terrain, but the code should
    be provided in uppercase.
    The function is designed to work with different localization codes, such as "PT"
    for Portugal or "EU" for standard CEN.
    The zones can include "ZonaA", "ZonaB", etc., and the terrain types can include
    "I", "II", "III", etc.
    This function is useful for engineers and designers who need to apply wind
    loads according to Eurocode standards.

    Args:
        code (str, optional): Localization code. Defaults to "PT".
        zone (str, optional): Wind zone. Defaults to "ZonaA" for Portugal.
        terrain (str, optional): terrain type. Defaults to "II".

    Returns:
        tuple: wind load parameters for the specified code, zone, and terrain.
        The tuple contains:
            - vb0 (float): base velocity for the specified zone.
            - zmin (float): minimum height for the specified terrain.
            - z0 (float): roughness length for the specified terrain.
            - z0II (float): roughness length for terrain II.

    """
    terrain = str.upper(terrain)
    code = str.upper(code)

    codes = WindLoads["locale"]
    if code not in codes:
        msg = f"Invalid code: {code}. Available codes: {list(codes)}"
        raise ValueError(msg)
    wind = codes[code]
    if zone not in wind["base_velocity"]:
        msg = (f"Invalid wind zone: {zone}. Available zones for {code}: "
               f"{list(wind['base_velocity'])}")
        raise ValueError(msg)
    if terrain not in wind["terrain"]:
        msg = (f"Invalid terrain: {terrain}. Available terrains for {code}: "
               f"{list(wind['terrain'])}")
        raise ValueError(msg)
    vb0 = wind["base_velocity"][zone]["vb0"]
    z0 = wind["terrain"][terrain]["z0"]
    z0II = wind["terrain"]["II"]["z0"]  # noqa: N806
    zmin = wind["terrain"][terrain]["zmin"]
    return vb0, zmin, z0, z0II
